    'ExecutionEnvironment', 'gpu');

%% Compute embeddings — correct method

%% Segment — imageSize is REQUIRED 3rd arg

    model, embeddings, imageSize, ...
    ForegroundPoints=fgPoints, ...
    BackgroundPoints=bgPoints);

%% 3D propagation: seed-and-propagate

    % Derive centroid from previous mask
    props = regionprops(prevMask, 'Centroid');

        emb, imageSize, ForegroundPoints=fg);
end

%% Save with spatial geometry preserved
segVol = medicalVolume(uint8(mask), V.VolumeGeometry);
segVol.Modality = 'SEG';
write(segVol, 'tumor_seg.nii.gz');

Wavelet FLAGSHIP EXAMPLE — MRI + CT Image Fusion

Fuse an MRI and CT image using wavelet-based fusion with proper coefficient-level rules: mean for approximation, max-absolute for details. Check decomposition level limits. Include quality metrics.
With Skill (180 lines)
%% Check decomposition level limits
maxLev = wmaxlev(size(img), waveletName);
decompLevel = min(decompLevel, maxLev);

%% Multi-level decomposition (wavedec2)
[C1, S1] = wavedec2(img1, decompLevel, 'sym8');
[C2, S2] = wavedec2(img2, decompLevel, 'sym8');

%% Fusion: mean approx + max-abs details
C_fused(1:approxLen) = 0.5*C1 + 0.5*C2;
for idx = (approxLen+1):length(C1)
    if abs(C2(idx)) > abs(C1(idx))
        C_fused(idx) = C2(idx);
    end
end

%% Energy-based detail selection per level
for lev = 1:decompLevel
    [cH1, cV1, cD1] = detcoef2('all', C1, S1, lev);
    e1 = conv2(sb.^2, kernel, 'same');
    mask = e1 >= e2;  % Local energy comparison
end

%% Quality: MI + spatial frequency + SSIM
mi = mutualInfo(img1, fused) + mutualInfo(img2, fused);
sf = spatialFrequency(fused);
s  = ssim(fused, img1);

Uses wmaxlev to validate decomposition level, wavedec2 for multi-level transform, detcoef2 to extract per-level subbands, im2double for correct [0,1] scaling. Two fusion strategies: max-mean and regional energy. Three quality metrics.

%% OD-space conversion (Beer-Lambert law)

%% Background masking (critical for PCA)

srcPixels = srcPixels(srcMask(:), :);

%% Macenko: PCA + percentile angle extraction
[srcCoeff, ~] = pca(srcPixels);
srcProj = srcPixels * srcCoeff(:, 1:2);
srcAngles = atan2(srcProj(:,2), srcProj(:,1));
srcMinAngle = prctile(srcAngles, 1);
srcMaxAngle = prctile(srcAngles, 99);

%% Stain separation (color deconvolution)
srcConc = (S'*S) \ (S' * srcAllPixels');

%% Normalize concentrations to reference
srcConc(ch,:) = srcC * (refP99 / srcP99);

%% Reconstruct via reference stain matrix
normalizedOD = refStainMatrix * srcConc;
normalizedImg = uint8(255 * 10.^(-normalizedOD));
